cmake_minimum_required(VERSION 3.26)
project(BaseWars LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release)
endif()

include(FetchContent)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
FetchContent_Declare(raylib GIT_REPOSITORY https://github.com/raysan5/raylib.git GIT_TAG 6.2 GIT_SHALLOW TRUE)
FetchContent_Declare(entt GIT_REPOSITORY https://github.com/skypjack/entt.git GIT_TAG v4.0.0 GIT_SHALLOW TRUE)
FetchContent_Declare(enet GIT_REPOSITORY https://github.com/lsalzman/enet.git GIT_TAG v1.3.18 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(raylib entt enet)

# ---- optional Steamworks SDK: drop it into third_party/steamworks (folder containing public/steam/steam_api.h)
set(STEAM_SDK_DIR "${CMAKE_SOURCE_DIR}/third_party/steamworks" CACHE PATH "Steamworks SDK root")
set(BW_STEAM OFF)
foreach(cand "${STEAM_SDK_DIR}" "${STEAM_SDK_DIR}/sdk")
    if(EXISTS "${cand}/public/steam/steam_api.h")
        set(STEAM_ROOT "${cand}")
        set(BW_STEAM ON)
    endif()
endforeach()
if(BW_STEAM)
    if(WIN32)
        set(STEAM_BIN_DIR "${STEAM_ROOT}/redistributable_bin/win64")
        set(STEAM_LIB_NAME steam_api64)
    elseif(APPLE)
        set(STEAM_BIN_DIR "${STEAM_ROOT}/redistributable_bin/osx")
        set(STEAM_LIB_NAME steam_api)
    else()
        set(STEAM_BIN_DIR "${STEAM_ROOT}/redistributable_bin/linux64")
        set(STEAM_LIB_NAME steam_api)
    endif()
    find_library(STEAM_API_LIB NAMES ${STEAM_LIB_NAME} PATHS "${STEAM_BIN_DIR}" NO_DEFAULT_PATH)
    if(NOT STEAM_API_LIB)
        message(WARNING "Steamworks SDK found but ${STEAM_LIB_NAME} is missing in ${STEAM_BIN_DIR}; building without Steam")
        set(BW_STEAM OFF)
    endif()
endif()
message(STATUS "Steamworks: ${BW_STEAM} (${STEAM_SDK_DIR})")

add_executable(server server.cpp)
add_executable(client client.cpp)
# headless protocol bots: every tools/bots/<name>.cpp becomes an executable <name> (used to verify server features without a window)
option(BW_BUILD_BOTS "Build the headless test bots in tools/bots" ON)
set(BOT_TARGETS "")
if(BW_BUILD_BOTS)
    file(GLOB BOT_SOURCES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/tools/bots/*.cpp")
    foreach(src ${BOT_SOURCES})
        get_filename_component(bot_name ${src} NAME_WE)
        add_executable(${bot_name} ${src})
        list(APPEND BOT_TARGETS ${bot_name})
    endforeach()
endif()
foreach(t server client ${BOT_TARGETS})
    if(NOT TARGET ${t})
        continue()
    endif()
    target_include_directories(${t} PRIVATE ${enet_SOURCE_DIR}/include ${CMAKE_SOURCE_DIR})
    target_link_libraries(${t} PRIVATE EnTT::EnTT enet)
    if(WIN32)
        target_link_libraries(${t} PRIVATE ws2_32 winmm)
    endif()
    if(BW_STEAM AND (t STREQUAL "server" OR t STREQUAL "client"))
        target_compile_definitions(${t} PRIVATE BW_STEAM)
        target_include_directories(${t} PRIVATE "${STEAM_ROOT}/public")
        target_link_libraries(${t} PRIVATE "${STEAM_API_LIB}")
        # the Steam runtime library and a test app id (480 = Spacewar) sit next to each executable
        add_custom_command(TARGET ${t} POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different "${STEAM_API_LIB}" "$<TARGET_FILE_DIR:${t}>"
            COMMAND ${CMAKE_COMMAND} -E echo 480 > "$<TARGET_FILE_DIR:${t}>/steam_appid.txt")
    endif()
endforeach()
target_link_libraries(client PRIVATE raylib)

file(WRITE ${CMAKE_BINARY_DIR}/mkcfg.cmake [=[
if(NOT EXISTS "${DST}")
  file(WRITE "${DST}" "server_ip=127.0.0.1\nserver_port=7777\n")
endif()
]=])
add_custom_command(TARGET client POST_BUILD
    COMMAND ${CMAKE_COMMAND} -DDST=$<TARGET_FILE_DIR:client>/config.txt -P ${CMAKE_BINARY_DIR}/mkcfg.cmake)
